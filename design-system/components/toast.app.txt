import React, { useEffect, useRef, useState } from "react";
import { AccessibilityInfo, Animated, Easing, Keyboard, StyleSheet, Text, View } from "react-native";
import { tokens } from "@/design-system/tokens"; // sync-tokens 로 만든 app 용 파일
import { Icon } from "./Icon"; // 앱의 Icon Library 부품

type Tone = "neutral" | "positive" | "critical" | "warning" | "informative";

type Props = {
  open: boolean;
  /** 시간이 지나 사라질 때 부른다. */
  onClose: () => void;
  /** 한 줄, 길어도 두 줄. */
  message: string;
  tone?: Tone;
  /** 보이는 시간(ms). 두 줄이면 늘리되 4000 을 넘기지 않는다. */
  duration?: number;
  /** 화면 아래에 주요 버튼이 있을 때. 버튼을 가리지 않게 그 위에 띄운다. */
  aboveAction?: boolean;
  /** 홈 표시 줄 높이. 화면에서 safe area 의 아래 값을 넘긴다. */
  bottomInset?: number;
};

const { color, dimension, radius, fontSize, fontWeight, lineHeight, iconSize, shadow } = tokens;

const MAX_DURATION = 4000;

// 면의 색은 같고 앞의 아이콘만 바뀐다. neutral 은 아이콘이 없다.
const ICONS = {
  positive: { name: "checkbox-circle-fill", color: color.fg.positive },
  critical: { name: "error-warning-fill", color: color.fg.critical },
  warning: { name: "alert-fill", color: color.fg.warning },
  informative: { name: "information-fill", color: color.fg.informative },
} as const;

// 루트 화면의 맨 위에 한 번만 둔다. Modal 로 띄우지 않으므로 토스트 밖은 그대로 누를 수 있다.
export function Toast({ open, onClose, message, tone = "neutral", duration = 2000, aboveAction = false, bottomInset = 0 }: Props) {
  // mounted: 화면에 있는지. 사라지는 움직임이 끝난 뒤에 뺀다.
  const [mounted, setMounted] = useState(open);
  const [reduceMotion, setReduceMotion] = useState(false);
  const [keyboard, setKeyboard] = useState(0);
  const progress = useRef(new Animated.Value(0)).current;
  const latestClose = useRef(onClose);
  latestClose.current = onClose;

  useEffect(() => {
    AccessibilityInfo.isReduceMotionEnabled().then(setReduceMotion);
    const subscription = AccessibilityInfo.addEventListener("reduceMotionChanged", setReduceMotion);
    return () => subscription.remove();
  }, []);

  // 키보드가 올라와 있으면 그 높이만큼 위로 올린다.
  useEffect(() => {
    const show = Keyboard.addListener("keyboardDidShow", (event) => setKeyboard(event.endCoordinates.height));
    const hide = Keyboard.addListener("keyboardDidHide", () => setKeyboard(0));
    return () => {
      show.remove();
      hide.remove();
    };
  }, []);

  useEffect(() => {
    if (open) setMounted(true);
    Animated.timing(progress, {
      toValue: open ? 1 : 0,
      duration: tokens.motion.duration.d3,
      easing: Easing.bezier(...(open ? tokens.motion.easing.enter : tokens.motion.easing.exit)),
      useNativeDriver: true,
    }).start(({ finished }) => {
      if (finished && !open) setMounted(false);
    });
  }, [open, progress]);

  // 정해진 시간이 지나면 스스로 닫는다. 문구가 바뀌면 시간을 다시 세고 새 문구를 읽어 준다.
  useEffect(() => {
    if (!open) return;
    AccessibilityInfo.announceForAccessibility(message);
    const timer = setTimeout(() => latestClose.current(), Math.min(duration, MAX_DURATION));
    return () => clearTimeout(timer);
  }, [open, message, duration]);

  if (!mounted) return null;

  const icon = tone === "neutral" ? null : ICONS[tone];
  // 버튼 위에 띄울 때는 버튼 영역(x16)에 x2 를 더한 만큼 올린다.
  const bottom = (aboveAction ? dimension.x16 + dimension.x2 : dimension.x6) + (keyboard > 0 ? keyboard : bottomInset);
  // 동작 줄이기에서는 자리 이동을 빼고 투명도만 바꾼다.
  const translateY = reduceMotion ? 0 : progress.interpolate({ inputRange: [0, 1], outputRange: [dimension.x2, 0] });

  return (
    <View pointerEvents="box-none" style={[styles.root, { bottom }]}>
      <Animated.View accessibilityLiveRegion="polite" style={[styles.surface, { opacity: progress, transform: [{ translateY }] }]}>
        {icon ? <Icon name={icon.name} size={iconSize.md} color={icon.color} /> : null}
        <Text numberOfLines={2} style={styles.message}>
          {message}
        </Text>
      </Animated.View>
    </View>
  );
}

const styles = StyleSheet.create({
  root: { position: "absolute", left: 0, right: 0, alignItems: "center", paddingHorizontal: dimension.x4 },
  surface: {
    flexDirection: "row",
    alignItems: "center",
    gap: dimension.x4,
    paddingVertical: dimension.x3,
    paddingHorizontal: dimension.x7,
    borderRadius: radius.full,
    borderWidth: 1,
    borderColor: color.stroke.neutralWeak,
    backgroundColor: color.bg.layerFloating,
    // boxShadow 는 RN 0.76 이상에서 쓴다. 그보다 낮으면 shadow* 와 elevation 으로 옮긴다.
    boxShadow: shadow.s2,
  },
  message: { flexShrink: 1, color: color.fg.neutral, fontSize: fontSize.t4, fontWeight: fontWeight.medium, lineHeight: fontSize.t4 * lineHeight.t4 },
});
