import React from "react";
import { Pressable, StyleSheet, Text, View } from "react-native";
import Svg, { Path } from "react-native-svg";
import { tokens } from "@/design-system/tokens"; // sync-tokens 로 만든 app 용 파일

type Props = {
  label: string;
  description?: string;
  checked: boolean;
  onCheckedChange?: (checked: boolean) => void;
  invalid?: boolean;
  disabled?: boolean;
};

const { color, dimension, radius, fontSize } = tokens;

export function Checkbox({ label, description, checked, onCheckedChange, invalid, disabled }: Props) {
  // 고르지 않은 네모의 테두리는 면과 3:1 이 되는 stroke.neutral.
  const borderColor = disabled ? color.stroke.neutralWeak : checked ? color.stroke.brandSolid : invalid ? color.fg.critical : color.stroke.neutral;
  const backgroundColor = disabled ? color.bg.disabled : checked ? color.bg.brandSolid : color.bg.layerDefault;

  return (
    <Pressable
      accessibilityRole="checkbox"
      accessibilityState={{ checked, disabled: !!disabled }}
      accessibilityLabel={label}
      accessibilityHint={description}
      disabled={disabled}
      onPress={() => onCheckedChange?.(!checked)}
      // 줄 전체가 눌린다. 눌리는 동안의 반응도 줄 전체의 배경으로 보여 준다.
      style={({ pressed }) => [styles.row, pressed && { backgroundColor: color.bg.neutralWeak }]}
    >
      {/* 높이를 레이블 한 줄과 같게 두어, 레이블이 여러 줄이어도 네모가 첫 줄에 맞는다. */}
      <View style={styles.line}>
        <View style={[styles.box, { borderColor, backgroundColor }]}>
          {/* 고른 상태는 색과 함께 체크 표시로 알린다. */}
          {checked ? (
            <Svg width={tokens.iconSize.sm} height={tokens.iconSize.sm} viewBox="0 0 24 24">
              <Path d="M9.55 17.3 4.6 12.35l1.4-1.4 3.55 3.55 8.45-8.45 1.4 1.4-9.85 9.85Z" fill={disabled ? color.fg.disabled : color.fg.onBrandSolid} />
            </Svg>
          ) : null}
        </View>
      </View>
      <View style={styles.text}>
        <Text style={{ color: disabled ? color.fg.disabled : color.fg.neutral, fontSize: fontSize.t5 }}>{label}</Text>
        {description ? <Text style={{ color: disabled ? color.fg.disabled : color.fg.neutralMuted, fontSize: fontSize.t3 }}>{description}</Text> : null}
      </View>
    </Pressable>
  );
}

const styles = StyleSheet.create({
  row: { minHeight: dimension.x12, paddingVertical: dimension.x3, flexDirection: "row", alignItems: "flex-start", gap: dimension.x3 },
  line: { height: dimension.x6, justifyContent: "center" },
  box: { width: dimension.x6, height: dimension.x6, borderRadius: radius.r1, borderWidth: 1, alignItems: "center", justifyContent: "center" },
  text: { flex: 1 },
});
